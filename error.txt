set -euo pipefail
OUT="results/spec-v1/shifted-profile-$(date +%Y%m%d-%H%M%S)-$$"
mkdir -p "$OUT"

for impl in deepgemm_tilelang ascendc_pre_plus_torch_apply; do
  python operators/spec_v1/bench.py \
    --suite mhc_shifted --tokens 8192 --profile "$impl" \
    --output "$OUT/$impl" \
    2>&1 | tee "$OUT/$impl.log"
done

find "$OUT" -type f -name kernel_details.csv \
  -print -exec cat {} \; | tee "$OUT/kernel-details.txt"
