cd /home/z30086261/Q4
BASE="results/mhc/ext-profile-$(date +%Y%m%d-%H%M%S)"

for case in "sinkhorn 512 4096" "sinkhorn 8192 4096" \
            "pre 8192 4096" "pre_sinkhorn 8192 4096"; do
  read -r suite tokens hidden <<< "$case"
  for impl in tilelang ascendc; do
    python operators/mhc/extended/bench_forward.py \
      --suite "$suite" --tokens "$tokens" --hidden "$hidden" \
      --profile "$impl" \
      --output "$BASE/$suite-$tokens-$hidden-$impl" \
      || break 2
  done
done

find "$BASE" -type f -name kernel_details.csv \
  -print -exec cat {} \; > mhc-extended-kernels.txt
