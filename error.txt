import json
from pathlib import Path

root = Path("reports/mha/latest")
files = sorted(root.glob("mha_cube_splitkv*.json"))
if not files:
    print("未找到报告，请确认在 FGW 仓库根目录运行。")

for p in files:
    r = json.loads(p.read_text())
    print("\nFILE:", p.name)
    for key in ("status", "phase", "passed", "total_checks",
                "correctness", "active_case", "active_variant"):
        if key in r:
            print(key, ":", r[key])

    errors = [x for x in r.get("results", [])
              if x.get("status") in ("error", "failed")]
    if errors:
        x = errors[0]
        print("失败样例:", {
            k: x.get(k) for k in
            ("case", "dtype", "causal", "pattern", "num_splits")
        })
    error = r.get("error") or next(
        (x.get("error") for x in errors if x.get("error")), None)
    if error:
        print("ERROR:\n", error[-8000:])

    if r.get("groups"):
        print("GROUPS:", json.dumps(r["groups"], ensure_ascii=False))

    for s in r.get("summary", []):
        print(s["key"],
              "配置:", s.get("cube_configuration"),
              "耗时/MFU:", s.get("metrics"),
              "对split1加速比:", s.get("speedup_split_over_split1"),
              "对AscendC加速比:", s.get("speedup_cube_over_ascendc"))
PY
