set -euo pipefail
OUT="results/spec-v1/mhc-perf-$(date +%Y%m%d-%H%M%S)-$$"
mkdir -p "$OUT"

for rep in 1 2 3; do
  for suite in mhc_post mhc_shifted; do
    for tokens in 512 8192; do
      tag="$suite-N$tokens-rep$rep"
      python operators/spec_v1/bench.py \
        --suite "$suite" --tokens "$tokens" \
        --warmup 100 --iters 100 --rounds 9 \
        --output "$OUT/$tag" \
        2>&1 | tee "$OUT/$tag.log"
    done
  done
done
