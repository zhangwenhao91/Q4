python - <<'PY'
import json
from pathlib import Path

root = Path("reports/mha/latest")
for name in (
    "mha_cube_ascendc_mfu_smoke_summary.json",
    "mha_cube_ascendc_mfu_summary.json",
):
    p = root / name
    if not p.exists():
        continue
    r = json.loads(p.read_text())
    print("\nFILE:", name)
    for key in ("status", "phase", "correctness", "active_case", "active_variant"):
        print(key, ":", r.get(key))
    if r.get("error"):
        print("ERROR:\n", r["error"][-8000:])
    print("GROUPS:", json.dumps(r.get("groups", {}), ensure_ascii=False))
    for s in r.get("summary", []):
        metrics = s.get("metrics", {})
        print(
            s["key"],
            "Cube:", metrics.get("cube_general"),
            "AscendC:", metrics.get("ascendc_fia_v2"),
            "speedup:", s.get("speedup_cube_over_ascendc"),
        )
PY
