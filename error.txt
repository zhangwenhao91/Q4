# 单独验证 TileLang Post 前向，暂不依赖 AscendC 包。
python - <<'PY' 2>&1 | tee mhc-tilelang-smoke.txt
import torch
import torch_npu
from tile_kernels.mhc import mhc_post_fwd

torch.npu.set_device(0)
torch.manual_seed(42)

B, S, N, D = 1, 512, 4, 4096
r = torch.randn(B, S, N, D, dtype=torch.bfloat16)
x = torch.randn(B, S, D, dtype=torch.bfloat16)
c = torch.randn(B, S, N, N).softmax(-1)
h = torch.rand(B, S, N, 1) * 2

ref = (
    torch.einsum("bsij,bsid->bsjd", c, r.float())
    + x.float().unsqueeze(-2) * h
).bfloat16()

with torch.inference_mode():
    out = mhc_post_fwd(x.npu(), r.npu(), h.npu(), c.npu())
    torch.npu.synchronize()
    out = out.cpu()

print("输出形状:", tuple(out.shape))
print("最大绝对误差:", (out.float() - ref.float()).abs().max().item())
torch.testing.assert_close(out, ref, rtol=0.02, atol=0.02)
print("TileLang mHC Post 前向冒烟验证通过")
PY
